# R packages behind src/shapes_analysis.R, as installed when this post was
# rendered (R 4.0.5, x86_64, Homebrew). Version pins only -- there is no R
# equivalent of the Python requirements.txt workflow here.
#
# rgl is a hard dependency of shapes (its NAMESPACE does `import(rgl)`), and it
# cannot be built from source on this machine: the C++ toolchain cannot find the
# libc++ headers, so `#include <algorithm>` fails for any package, not just this
# one. Both came from the prebuilt CRAN macOS binaries for R 4.0 instead:
#
#   B=https://cran.r-project.org/bin/macosx/contrib/4.0
#   curl -sO $B/rgl_0.108.3.tgz && curl -sO $B/shapes_1.2.6.tgz
#   Rscript -e 'install.packages(c("rgl_0.108.3.tgz","shapes_1.2.6.tgz"),
#                                repos=NULL, type="mac.binary")'
#
# The rest resolve from source normally (they are pure R):
#
#   Rscript -e 'install.packages(c("jsonlite","minpack.lm","scatterplot3d",
#                                  "fitdistrplus","htmlwidgets","htmltools",
#                                  "knitr","magrittr","R6","crosstalk",
#                                  "base64enc","colorRamps"))'
#
# src/warp_movie.R additionally needs jpeg, png and gifski, all binaries:
#
#   for f in jpeg_0.1-9 png_0.1-7 gifski_1.6.6-1; do curl -sO $B/$f.tgz; done
#   Rscript -e 'install.packages(c("jpeg_0.1-9.tgz","png_0.1-7.tgz",
#                                  "gifski_1.6.6-1.tgz"), repos=NULL,
#                                type="mac.binary")'
#
# Morpho is deliberately NOT used, though it would have supplied tps2d. Its CRAN
# binary links /Library/Frameworks/R.framework/.../libRblas.dylib, and this
# Homebrew R ships libRlapack.dylib but no libRblas.dylib, so the package
# installs and then fails at dlopen. Same for RcppArmadillo, which it needs. The
# thin-plate spline in warp_movie.R is written out longhand instead.
#
# Optimising the GIF also uses the gifsicle CLI (brew install gifsicle); the
# script skips that step if it is not on PATH.
